    <script>
        const populations = {
            global: { count: 8118835421, rate: 2.3 },
            europe: { count: 748935267, rate: 0.08 },
            uk: { count: 67886011, rate: 0.02 },
            netherlands: { count: 17564014, rate: 0.004 },
            haarlem: { count: 162543, rate: 0.0003 }
        };
        
        function formatNumber(num) {
            return num.toLocaleString();
        }
        
        function updateCounters() {
            Object.keys(populations).forEach(region => {
                const data = populations[region];
                data.count += data.rate;
                const element = document.getElementById(`${region}-count`);
                if (element) {
                    element.textContent = formatNumber(Math.floor(data.count));
                    console.log(`Updating ${region}: ${element.textContent}`);
                } else {
                    console.error(`Element ${region}-count not found`);
                }
            });
        }
        
        setInterval(updateCounters, 1000);
        
        document.addEventListener('DOMContentLoaded', () => {
            Object.keys(populations).forEach(region => {
                const element = document.getElementById(`${region}-count`);
                if (element) {
                    element.textContent = formatNumber(Math.floor(populations[region].count));
                    console.log(`Initial ${region}: ${element.textContent}`);
                } else {
                    console.error(`Initial element ${region}-count not found`);
                }
            });
        });
